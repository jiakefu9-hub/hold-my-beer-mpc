cmake_minimum_required(VERSION 3.16)
project(g1_arm_delay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(MUJOCO_ROOT "" CACHE PATH "Installed MuJoCo Python package directory")
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS "${MUJOCO_ROOT}/include" REQUIRED)
find_library(MUJOCO_LIBRARY NAMES mujoco libmujoco.so.3.6.0 HINTS "${MUJOCO_ROOT}" NO_DEFAULT_PATH REQUIRED)
add_library(g1_arm_delay SHARED delay.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/delay.cpp" G1_DELAY_SOURCE_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS delay.cpp)
target_compile_definitions(g1_arm_delay PRIVATE G1_DELAY_SOURCE_SHA256="${G1_DELAY_SOURCE_SHA256}")
target_include_directories(g1_arm_delay PRIVATE "${MUJOCO_INCLUDE_DIR}")
target_link_libraries(g1_arm_delay PRIVATE "${MUJOCO_LIBRARY}")
target_compile_options(g1_arm_delay PRIVATE -O3 -Wall -Wextra -Wpedantic)
set_target_properties(g1_arm_delay PROPERTIES BUILD_RPATH "${MUJOCO_ROOT}")
